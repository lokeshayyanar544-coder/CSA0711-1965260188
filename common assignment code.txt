import socket
import threading

HOST = "127.0.0.1"
TCP_PORT = 5000
UDP_PORT = 5001
BUFFER = 4096

def handle_tcp(conn, addr):
    try:
        while True:
            data = conn.recv(BUFFER)
            if not data:
                break
            text = data.decode()
            client_id, _, message = text.partition("|")
            reply = f"ACK|{client_id}|{len(data)}"
            conn.sendall(reply.encode())
            print("TCP received:", text)
    finally:
        conn.close()

def tcp_server():
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    s.bind((HOST, TCP_PORT))
    s.listen()
    print("TCP server listening on", TCP_PORT)
    while True:
        conn, addr = s.accept()
        threading.Thread(
            target=handle_tcp, args=(conn, addr), daemon=True
        ).start()

def udp_server():
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    s.bind((HOST, UDP_PORT))
    print("UDP server listening on", UDP_PORT)
    while True:
        data, addr = s.recvfrom(BUFFER)
        text = data.decode()
        client_id, _, message = text.partition("|")
        reply = f"ACK|{client_id}|{len(data)}"
        s.sendto(reply.encode(), addr)
        print("UDP received:", text)

if __name__ == "__main__":
    threading.Thread(target=tcp_server, daemon=True).start()
    threading.Thread(target=udp_server, daemon=True).start()
    input("Server running. Press Enter to stop...")



import socket
import sys
import time

def tcp_client(cid, ip, port, message):
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    start = time.perf_counter()
    s.connect((ip, port))
    print("Connection Status: CONNECTED")
    s.sendall(f"{cid}|{message}".encode())
    ack = s.recv(1024).decode()
    elapsed = (time.perf_counter() - start) * 1000
    print("Message Received:", ack)
    print(f"Packet Transmission Time: {elapsed:.3f} ms")
    print("Acknowledgment Status:", "ACK RECEIVED" if ack.startswith("ACK") else "NO ACK")
    s.close()
    print("Connection Status: CLOSED")

def udp_client(cid, ip, port, message):
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    s.settimeout(2)
    start = time.perf_counter()
    s.sendto(f"{cid}|{message}".encode(), (ip, port))
    print("Connection Status: CONNECTIONLESS")
    try:
        data, _ = s.recvfrom(1024)
        elapsed = (time.perf_counter() - start) * 1000
        print("Message Received:", data.decode())
        print(f"Packet Transmission Time: {elapsed:.3f} ms")
        print("Acknowledgment Status: ACK RECEIVED")
    except socket.timeout:
        print("Acknowledgment Status: TIMEOUT")
    s.close()

if __name__ == "__main__":
    cid, ip, port, protocol, message = sys.argv[1:6]
    port = int(port)
    if protocol.upper() == "TCP":
        tcp_client(cid, ip, port, message)
    elif protocol.upper() == "UDP":
        udp_client(cid, ip, port, message)
    else:
        print("Choose TCP or UDP")
